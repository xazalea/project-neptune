
      async function boot() {
        // Register bridge message listener exactly once
        window.addEventListener('message', handleIframeMessage);
        setupDragDrop();
        updateQuickLinks();

        // Parse URL params
        var params = new URLSearchParams(window.location.search);
        var target = params.get('url') || params.get('') || null;
        var snapshot = params.get('state');
        var proxyMode = params.has('__nptn');

        if (proxyMode) {
          landing.style.display = 'none';
          setBootStatus('Proxy mode — skipping boot', true);
          return;
        }

        // Direct proxy entrypoint: neptune.svg?url=https://proxied.site
        if (target) {
          landing.style.display = 'none';
          document.getElementById('boot-status').textContent = 'Loading ' + target.replace(/^https?:\/\//, '').split('/')[0] + '...';
        }

        window.bootExample = function(url) {
          document.getElementById('landing-url').value = url;
          window.bootFromLanding();
        };

        log(L.INF, '========================================');
        log(L.INF, 'project: neptune v' + APP_VERSION);
        log(L.INF, '========================================');
        if (target) log(L.INF, 'Target: ' + target);
        if (snapshot) log(L.INF, 'Snapshot: ' + atob(snapshot).length + ' bytes');

        // ── Try NeptuneBoot v3 modular bootloader ──
        var useBootloader = false;
        try {
          if (BOOTLOADER_B64 && BOOTLOADER_B64.length > 0 && BOOTLOADER_B64.indexOf('{{') === -1) {
            eval(atob(BOOTLOADER_B64));
            if (window.NeptuneBoot) {
              var artifacts = {
                swCode: (SW_CODE_B64 && SW_CODE_B64.indexOf('{{') === -1) ? SW_CODE_B64 : null,
                wasmJs: (WASM_JS_B64 && WASM_JS_B64.indexOf('{{') === -1) ? WASM_JS_B64 : null,
                wasmBin: (WASM_BIN_B64 && WASM_BIN_B64.indexOf('{{') === -1) ? WASM_BIN_B64 : null,
                networkAdapter: (NETWORK_ADAPTER_B64 && NETWORK_ADAPTER_B64.indexOf('{{') === -1) ? NETWORK_ADAPTER_B64 : null,
                fingerprintJs: (FINGERPRINT_JS_B64 && FINGERPRINT_JS_B64.indexOf('{{') === -1) ? FINGERPRINT_JS_B64 : null,
                transportDetector: (TRANSPORT_DETECTOR_B64 && TRANSPORT_DETECTOR_B64.indexOf('{{') === -1) ? TRANSPORT_DETECTOR_B64 : null,
                transportModules: {},
                engineModules: {},
                aiEngine: (ENGINE_AI_B64 && ENGINE_AI_B64.length > 0 && ENGINE_AI_B64.indexOf('{{') === -1) ? ENGINE_AI_B64 : null,
                uiModules: {},
              };
              function _addModule(obj, key, b64) { if (b64 && b64.length > 0 && b64.indexOf('{{') === -1) obj[key] = b64; }
              var tm = artifacts.transportModules;
              _addModule(tm, 'transport.js', TRANSPORT_BASE_B64);
              _addModule(tm, 'turn.js', TRANSPORT_TURN_B64);
              _addModule(tm, 'webrtc.js', TRANSPORT_WEBRTC_B64);
              _addModule(tm, 'smoltcp.js', TRANSPORT_SMOLTCP_JS_B64);
              _addModule(tm, 'houdini.js', TRANSPORT_HOUDINI_B64);
              var em = artifacts.engineModules;
              _addModule(em, 'rewriter.js', ENGINE_REWRITER_B64);
              _addModule(em, 'security.js', ENGINE_SECURITY_B64);
              _addModule(em, 'tracker.js', ENGINE_TRACKER_B64);
              _addModule(em, 'cookies.js', ENGINE_COOKIES_B64);
              _addModule(em, 'cache.js', ENGINE_CACHE_B64);
              _addModule(em, 'obfuscator.js', ENGINE_OBFUSCATOR_B64);
              _addModule(em, 'logger.js', ENGINE_LOGGER_B64);
              useBootloader = true;
              log(L.INF, 'NeptuneBoot v3 ready — ' + Object.keys(tm).length + ' transports, ' + Object.keys(em).length + ' engines');
            }
          }
        } catch(e) { log(L.WRN, 'Bootloader init: ' + e.message); }

        if (useBootloader) {
          var _dom = {
            bootOverlay: document.getElementById('boot-overlay'),
            bootStatus: document.getElementById('boot-status'),
          };
          var _cfg = Object.assign({}, storage.settings);
          if (fingerprintEngine) _cfg.fingerprintCode = fingerprintEngine;

          NeptuneBoot.boot({
            artifacts: artifacts,
            dom: _dom,
            config: _cfg,
            onReady: function(bootResult, transports, params, config) {
              if (window.__nptn_kernel) kernel = window.__nptn_kernel;
              if (window.__nptn_netStack) netStack = window.__nptn_netStack;
              if (window.__nptn_networkAdapter) netAdapter = window.__nptn_networkAdapter;
              if (window.__nptn_fingerprintCode) fingerprintEngine = window.__nptn_fingerprintCode;
              if (window.__nptn_aiEngine) aiEngine = window.__nptn_aiEngine;
              if (navigator.serviceWorker.controller) { swReady = true; navigator.serviceWorker.addEventListener('message', handleSWMessage); }

              syncConfigToSW();

              if (transports && transports.best) {
                document.getElementById('status-strategy').textContent = 'Transport: ' + transports.best.name;
              }

              if (bootResult.success && target) {
                if (navigator.serviceWorker.controller) {
                  navigator.serviceWorker.controller.postMessage({ type: 'SET_TARGET', url: target, snapshot: snapshot || null });
                }
                // Direct proxy entrypoint — immediately load target, no landing flash
                createTab(target);
                // Ensure chrome shows the proxied URL
                urlBar.value = target;
                urlBar.style.color = '#00ff88';
              }

              setInterval(function() {
                if (navigator.serviceWorker.controller) {
                  navigator.serviceWorker.controller.postMessage({ type: 'GET_STATS' });
                }
              }, 2000);

              if (netAdapter) { try { netAdapter.start(); } catch(e) {} }
              toast('Neptune v3 ready — ' + (bootResult.totalMs || '?') + 'ms');
            }
          });

        } else {
          // Legacy v2 boot (fallback)
          setBootStatus('Loading WASM kernel...');
          var wasmOk = await loadWASM();
          if (!wasmOk) { setBootStatus('WASM unavailable — proxy still functional', true); }

          setBootStatus('Registering Service Worker...');
          var ok = await bootSW();
          if (!ok) {
            setBootStatus('Service Worker failed — proxy requires SW', true);
            toast('Service Worker failed. Proxy cannot function.', 'error');
            return;
          }

          setBootStatus('Initializing network stack...');
          loadNetworkAdapter(); loadFingerprintEngine();
          if (netAdapter) {
            netAdapter.start();
            log(L.INF, 'NetworkAdapter started — smoltcp fetch()-based networking active');
            toast('Network stack online', 'info');
          }

          setBootStatus('Initializing persistence engine...');
          try {
            if (ENGINE_PERSISTENCE_B64 && ENGINE_PERSISTENCE_B64.indexOf('{{') !== 0) {
              eval(atob(ENGINE_PERSISTENCE_B64));
              if (typeof NeptunePersistence !== 'undefined') {
                window.__nptn_persist = new NeptunePersistence.NeptunePersistence();
                window.__nptn_persist.init().then(function() {
                  log(L.INF, 'Persistence engine initialized');
                  toast('Persistence: wake lock + background sync active', 'info');
                }).catch(function(e) { log(L.WRN, 'Persistence init: ' + e.message); });
              }
            }
          } catch(e) { log(L.WRN, 'Persistence engine failed: ' + e.message); }

          setBootStatus('Initializing mesh engine...');
          try {
            if (ENGINE_MESH_B64 && ENGINE_MESH_B64.indexOf('{{') !== 0) {
              eval(atob(ENGINE_MESH_B64));
              if (typeof NeptuneMeshEngine !== 'undefined') {
                window.__nptn_mesh = new NeptuneMeshEngine.NeptuneMeshEngine();
                window.__nptn_mesh.init().then(function() {
                  log(L.INF, 'Mesh engine initialized');
                  toast('Mesh: multi-tab coordination active', 'info');
                }).catch(function(e) { log(L.WRN, 'Mesh init: ' + e.message); });
              }
            }
          } catch(e) { log(L.WRN, 'Mesh engine failed: ' + e.message); }

          setBootStatus('Ready', true);

          if (ok && target) {
            if (navigator.serviceWorker.controller) {
              navigator.serviceWorker.controller.postMessage({ type: 'SET_TARGET', url: target, snapshot: snapshot || null });
            }
            createTab(target);
          }

          setInterval(function() {
            if (navigator.serviceWorker.controller) {
              navigator.serviceWorker.controller.postMessage({ type: 'GET_STATS' });
            }
          }, 2000);
        }
      }

      boot();

      // Expose functions to window for onclick handlers
      window.navigateTo = navigateTo;
      window.deleteBookmark = deleteBookmark;
      window.clearHistory = clearHistory;
      window.toggleSetting = toggleSetting;
      window.saveSettings = saveSettings;
      window.resetSettings = resetSettings;
      window.clearAllData = clearAllData;
      window.clearNetworkLog = clearNetworkLog;
      window.exportNetworkLog = exportNetworkLog;

      window.toolUrlEncode = toolUrlEncode;
      window.toolUrlDecode = toolUrlDecode;
      window.toolB64Encode = toolB64Encode;
      window.toolB64Decode = toolB64Decode;
      window.toolJsonFormat = toolJsonFormat;
      window.exportAllData = exportAllData;
      window.importAllData = importAllData;
    })();
