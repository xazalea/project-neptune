      }

      // ====== SERVICE WORKER ======
      async function bootSW() {
        if (!('serviceWorker' in navigator)) { toast('Service Workers not supported', 'error'); return false; }
        try {
          // ServiceWorker registration from './sw.js' — the SW file must be deployed
          // alongside neptune.svg on the same CDN (e.g., jsDelivr serves both from GitHub).
          var swURL = './sw.js';
          log(L.INF, 'Registering SW from: ' + swURL);
          var reg = await navigator.serviceWorker.register(swURL, { scope: './', updateViaCache: 'none' });
          log(L.INF, 'SW scope: ' + reg.scope);

          await new Promise(function(r){
            var sw = reg.installing || reg.waiting || reg.active;
            if (!sw || sw.state === 'activated') { r(); return; }
            sw.addEventListener('statechange', function(e){ log(L.INF, 'SW state: ' + e.target.state); if (e.target.state === 'activated') r(); });
          });

          if (!navigator.serviceWorker.controller) {
            await new Promise(function(r){ navigator.serviceWorker.addEventListener('controllerchange', r, {once:true}); });
          }

          navigator.serviceWorker.addEventListener('message', handleSWMessage);
          swReady = true;
          log(L.INF, 'SW controller active');

          // Sync config
          syncConfigToSW();
          return true;
        } catch(e) {
          toast('SW failed: ' + e.message, 'error');
          return false;
        }
      }

      function handleSWMessage(e) {
        var d = e.data;
        if (!d || !d.type) return;
        switch(d.type) {
          case 'STRATEGY_SELECTED':
            document.getElementById('status-strategy').textContent = 'Strategy: ' + d.strategy + (d.cors ? ' (CORS)' : '');
            urlStrategy.style.display = 'inline';
            urlStrategy.textContent = d.strategy.toUpperCase();
            urlStrategy.style.color = d.cors ? '#00ff88' : '#ffaa00';
            toast('Strategy: ' + d.strategy + (d.cors ? ' ✓' : ''));
            break;
          case 'REQUEST_LOG_ENTRY':
            addNetworkEntry(d.entry);
            break;
          case 'DOWNLOAD_DETECTED':
            downloads.unshift(d);
            if (activePanel === 'downloads') renderDownloadsPanel();
            toast('Download detected: ' + d.filename);
            break;
          case 'CONFIG_ACK':
            swConfig = d.config;
            break;
          case 'STATS_DATA':
            document.getElementById('status-requests').textContent = 'Requests: ' + d.totalRequests;
            document.getElementById('status-bandwidth').textContent = 'Bandwidth: ' + formatBytes(d.totalBytes);
            document.getElementById('status-blocked').textContent = 'Blocked: ' + d.blockedCount;
            break;
        }
      }

            else val = 'https://duckduckgo.com/html/?q=' + encodeURIComponent(val);
          }
          navigateTo(val);
        }
      });

      // ====== KEYBOARD SHORTCUTS ======
      document.addEventListener('keydown', function(e) {
        // Ctrl+L → focus URL bar
        if (e.ctrlKey && e.key === 'l') { e.preventDefault(); urlBar.focus(); urlBar.select(); }
        // F5 → refresh
        if (e.key === 'F5') { e.preventDefault(); refresh(); }
        // Ctrl+R → refresh
        if (e.ctrlKey && e.key === 'r') { e.preventDefault(); refresh(); }
        // Ctrl+T → new tab
        if (e.ctrlKey && e.key === 't') { e.preventDefault(); createTab(); }
        // Ctrl+W → close tab
        if (e.ctrlKey && e.key === 'w') { e.preventDefault(); var tab = tabs.find(function(t){return t.id===activeTabId}); if(tab)closeTab(tab.id); }
        // Ctrl+Shift+O → reopen closed tab (avoids browser's Ctrl+Shift+T conflict)
        if (e.ctrlKey && e.shiftKey && e.key === 'o') { e.preventDefault(); undoCloseTab(); }
        // Ctrl+PageUp / Ctrl+PageDown → switch Neptune tabs (avoid Ctrl+Tab browser conflict)
        if (e.ctrlKey && e.key === 'PageUp') { e.preventDefault(); var idx = tabs.findIndex(function(t){return t.id===activeTabId}); if(idx>0)activateTab(tabs[idx-1].id); }
        if (e.ctrlKey && e.key === 'PageDown') { e.preventDefault(); var idx = tabs.findIndex(function(t){return t.id===activeTabId}); if(idx<tabs.length-1)activateTab(tabs[idx+1].id); }
        // Alt+Left/Right → back/forward
        if (e.altKey && e.key === 'ArrowLeft') { e.preventDefault(); goBack(); }
        if (e.altKey && e.key === 'ArrowRight') { e.preventDefault(); goForward(); }
        // Ctrl+D → bookmark
        if (e.ctrlKey && e.key === 'd') { e.preventDefault(); addBookmark(); }
        // F11 → fullscreen
        if (e.key === 'F11') { e.preventDefault(); toggleFullscreen(); }
        // Escape → close panels
        if (e.key === 'Escape') {
          if (panelOverlay.style.display === 'flex') closePanel();
          else if (readerOverlay.style.display === 'flex') readerOverlay.style.display = 'none';
          else if (responsiveBar.style.display === 'flex') { responsiveBar.style.display = 'none'; setViewportSize('100%','100%'); }
        }
        // Ctrl+Shift+I → network
        if (e.ctrlKey && e.shiftKey && e.key === 'I') { e.preventDefault(); showPanel('network'); }
        // Ctrl+Shift+J → console
        if (e.ctrlKey && e.shiftKey && e.key === 'J') { e.preventDefault(); showPanel('console'); }
        // Ctrl+Shift+R → reader
        if (e.ctrlKey && e.shiftKey && e.key === 'R') { e.preventDefault(); toggleReaderMode(); }
        // Ctrl+Shift+M → responsive
        if (e.ctrlKey && e.shiftKey && e.key === 'M') { e.preventDefault(); toggleResponsive(); }
        // Ctrl+Shift+S → screenshot
        if (e.ctrlKey && e.shiftKey && e.key === 'S') { e.preventDefault(); takeScreenshot(); }
        // Ctrl+, → settings
        if (e.ctrlKey && e.key === ',') { e.preventDefault(); showPanel('settings'); }
        // Ctrl+H → history
        if (e.ctrlKey && e.key === 'h') { e.preventDefault(); showPanel('history'); }
        // Ctrl+B → bookmarks
        if (e.ctrlKey && e.key === 'b') { e.preventDefault(); showPanel('bookmarks'); }
        // Ctrl+F → find
        if (e.ctrlKey && e.key === 'f') { e.preventDefault(); findInPage(); }
        // Ctrl++ → zoom in
        if (e.ctrlKey && e.key === '=') { e.preventDefault(); zoomIn(); }
        // Ctrl+- → zoom out
        if (e.ctrlKey && e.key === '-') { e.preventDefault(); zoomOut(); }
        // Ctrl+0 → zoom reset
        if (e.ctrlKey && e.key === '0') { e.preventDefault(); zoomReset(); }
      });

      // ====== EXPORT STATE ======
      window.exportState = function() {
        if (!kernel) { toast('WASM not loaded', 'error'); return null; }
        if (!kernel.create_snapshot) { toast('Snapshot not supported by WASM build', 'warn'); return null; }
        try {
          var snap = kernel.create_snapshot();
          var u = new URL(window.location.href);
          u.searchParams.set('state', snap);
          log(L.INF, 'Snapshot: ' + snap.length + ' chars');
          return u.toString();
        } catch(e) { log(L.ERR, 'Export failed: ' + e.message); return null; }
      };
